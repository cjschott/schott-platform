#!/usr/bin/env bash
# The CSEL-000005 PRODUCTION FREEZE, exactly as it is to be typed.
#
# This file is reference text, not a script to run blind: the operator pastes it
# into a root-capable shell so every command and every refusal is visible on the
# console.
#
# WHAT THIS DOES. It puts one reviewed body under /etc/kyri/fabric and asks the
# released engine what it WOULD do with it. It does not write
# /var/lib/kyri/fabric, and it proves that afterwards.
#
# STEP 4 OF 4, THE LAST ONE. CADV-000008, CINST-000007 and CROUTE-0007 are
# written and accepted. This freezes the selection that binds the renewed route
# to the renewed admission, which is what `verify_selected_evidence` asks for and
# what the Stage-3 gate matrix needs to stop masking.
#
# =====================================================================
# WHAT THE ENGINE ENFORCES, AND WHAT IT DOES NOT -- MEASURED
# =====================================================================
#
# G11-BC-AS probed the released `select` against a copy of production with eight
# instant pairs. The result, precisely:
#
#   WHAT THE ENGINE DOES. `evaluated_at` is the instant eligibility is judged at.
#   An `evaluated_at` outside the admission window makes the selection resolve to
#   `selected_instance_id: null`.
#
#   WHAT THE ENGINE DOES NOT DO. It never REFUSES for any of it. In all eight
#   probes -- including a selection dated a day before the route it resolves
#   through, one dated after both windows closed, and one dated in December --
#   the engine returned `would_accept: true`. A null resolution is reported, not
#   rejected: the engine will persist an accepted selection that selects nothing.
#
#   AND `recorded_at` HAS NO EFFECT ON RESOLUTION AT ALL. A selection recorded
#   one second before, or seven hours before, the route it resolves through still
#   resolved to CINST-000007. Only `evaluated_at` moves the answer.
#
# THEREFORE TWO GATES BELOW ARE CEREMONY AUTHORITY AND MUST NOT BE REMOVED:
#
#   GATE 5  the ordering -- `recorded_at` at or after CROUTE-0007's own
#           `recorded_at`, before both expiries, and not in the future. The
#           engine checks none of this.
#
#   GATE 6  `selected_instance_id == CINST-000007`. NOT `would_accept`. This is
#           the single most important line in this file: `would_accept: true`
#           with a null selection is precisely the quiet failure the whole
#           renewal chain exists to repair, and it is what production would have
#           recorded if the route had been left alone.
#
# THE OPERATOR ASSERTS THE DECISION INSTANT. This block refuses without it:
#
#     SELECTION_DECISION_AT=2026-10-08T03:32:26-05:00 bash <this block>
#
# A selection is a DECISION; the instant is inside the reviewed body and every
# dependent digest derives from it. A mismatch means the body is RE-PREPARED, not
# edited.
#
# NOTHING HERE READS A CLOCK TO FILL IN A VALUE. `date` appears only in gates
# that REFUSE and in the eligibility and resolution probes, which must judge at
# the current instant to mean anything.
#
# =====================================================================
# `selected_at` IS DERIVED, NOT SUPPLIED
# =====================================================================
#
# The reviewed body carries `recorded_at` and `evaluated_at`. It does NOT carry
# `selected_at`: the released engine sets `selected_at = recorded_at` when it
# writes the record (`tools/fabric/selection.py`), and the CLI splats the body as
# keyword arguments, so a body carrying `selected_at` makes the engine refuse
# outright with "the decision body does not match this operation" -- which
# G11-BC-AS confirmed by trying it.
#
# So the operator's single decision instant lands in all three places by
# construction: `recorded_at`, `evaluated_at`, and the derived `selected_at`.
# Gate 4 checks the two it supplies; gate 7 checks the derived one on the copy.
#
# SELECTIONS DO NOT SUPERSEDE. Unlike advertisements, admissions and routes, no
# selection record carries a `supersedes` field -- all four accepted selections
# are "heads" if you scan for one. So there is no supersession head to check
# here, and this block does not pretend otherwise: it gates on the SEQUENCE being
# 4, on CSEL-000005 being absent, and on CSEL-000004 being byte-identical.
# Inventing a head check for a record kind that has no supersession would be
# asserting something the store cannot answer.
#
# THE STALE G11-BC-AM SELECTION BODY IS 606 BYTES, THE SAME AS THIS ONE -- the
# third same-length collision in this chain. A byte count cannot tell them apart,
# so it is refused BY DIGEST below.
#
# NO AUTHORITY IS BROADENED. capability CAPDEF-0001, contract CCON-0001, version
# 1.0.0, classification `internal`, locality `local-only` and node HOST-0001 are
# all equal to CSEL-000004, field by field. The package CPKG-0001, the host
# CHOST-0001, the architecture x86-64 and the operation `execute` are ADMISSION
# fields, carried by CINST-000007 -- which gate 2 pins by digest -- and a
# selection neither widens nor narrows them.
set -Eeuo pipefail

bash <<'FREEZE'
set -Eeuo pipefail

REVIEWED_REQUEST_ID=g11bcas-select-capdef-0001-ccon-0001-internal-local-only-host-0001-croute-0007

REVIEWED_RECORDED_AT=2026-10-08T03:32:26-05:00
REQUIRED_SELECTED_INSTANCE=CINST-000007

# The POST-CROUTE baselines. 1dc83d01 is the pre-route store and must never gate
# a CSEL step: the route write moved it.
FABRIC_BEFORE=758c6c6578abdfccc7d29d2f1b7e9d56cbf4d933636853c884fb599ef70862ba

# The governing chain, as production persisted it.

aggregate() { find "$1" -type f -print0 | sort -z | xargs -0 sha256sum | sha256sum | cut -d' ' -f1; }

# ---- GATE 0: THE OPERATOR SUPPLIES THE SELECTION DECISION INSTANT --------
printf -- '\n--- the selection decision instant, supplied by the operator ---\n'
if [[ -z "${SELECTION_DECISION_AT:-}" ]]; then
  echo "REFUSE: no SELECTION_DECISION_AT was supplied."
  echo "        Re-run as:  SELECTION_DECISION_AT=${REVIEWED_RECORDED_AT} bash <this block>"
  echo "        A selection is your decision, and the instant is yours to assert."
  exit 1
fi
test "${SELECTION_DECISION_AT}" = "${REVIEWED_RECORDED_AT}" || {
  echo "REFUSE: you supplied ${SELECTION_DECISION_AT}; the reviewed bytes carry ${REVIEWED_RECORDED_AT}."
  echo "        If your selection decision is a different instant, the body must be"
  echo "        RE-PREPARED -- the instant is inside it and its digest derives from"
  echo "        it. Do not edit the body here."
  exit 1; }

  "${SELECTION_DECISION_AT}"

# ---- GATE 1: THE POST-CROUTE PRODUCTION BASELINES ------------------------

for triple in "fabric:/var/lib/kyri/fabric:${FABRIC_BEFORE}" \
              "runtime:/data/kyri/capability-runtime:${RUNTIME_BEFORE}" \
              "trust:/var/lib/kyri/trust:${TRUST_BEFORE}"; do

  got="$(aggregate "${path}")"
  test "${got}" = "${want}" \
    || { echo "REFUSE: the ${label} store is ${got}, expected ${want}"; exit 1; }
  printf 'ok  %-8s %s\n' "${label}" "${got}"

# ---- GATE 2: THE WHOLE GOVERNING CHAIN, AS WRITTEN ----------------------
printf -- '\n--- the governing chain: advertisement, admission, route ---\n'
for triple in \
  "CADV-000008:/var/lib/kyri/fabric/capability-advertisements/CADV-000008.yaml:${CADV8_RECORD}" \
  "CINST-000007:/var/lib/kyri/fabric/capability-instances/CINST-000007.yaml:${CINST7_RECORD}" \
  "CROUTE-0007:/var/lib/kyri/fabric/capability-routes/CROUTE-0007.yaml:${CROUTE7_RECORD}"; do
  IFS=: read -r label path want <<<"${triple}"
  got="$(sha256sum "${path}" | cut -d' ' -f1)"
  test "${got}" = "${want}" \
    || { echo "REFUSE: ${label} is ${got}, not the record the operator wrote (${want})"; exit 1; }
  printf 'ok  %-13s %s\n' "${label}" "${got}"

# Each of the three must still be the head of its own kind, by scan.
for pair in "capability-advertisements:CADV-000008" \
            "capability-instances:CINST-000007" \
            "capability-routes:CROUTE-0007"; do
  IFS=: read -r dir id <<<"${pair}"
  test -z "$(grep -l "^supersedes: ${id}\$" \
               "/var/lib/kyri/fabric/${dir}"/*.yaml 2>/dev/null)" \
    || { echo "REFUSE: ${id} has already been superseded"; exit 1; }
done
test "$(sed -n 's/^lifecycle_state: //p' \
          /var/lib/kyri/fabric/capability-instances/CINST-000007.yaml)" = "admitted" \
  || { echo "REFUSE: CINST-000007 is not in the admitted lifecycle state"; exit 1; }
for pair in "capability-advertisement:8" "capability-instance:7" "capability-route:7"; do
  IFS=: read -r seq want <<<"${pair}"
  test "$(cat "/var/lib/kyri/fabric/sequences/${seq}.seq")" = "${want}" \
    || { echo "REFUSE: the ${seq} sequence is not ${want}"; exit 1; }

# And the route must actually offer the instance this selection must resolve to.
ROUTE_CANDIDATES="$(sed -n '/^candidate_instances:/,/^capability_id:/p' \

  || { echo "REFUSE: CROUTE-0007 offers ${ROUTE_CANDIDATES}, not ${REQUIRED_SELECTED_INSTANCE}"; exit 1; }

  "${ROUTE_CANDIDATES}"

# ---- GATE 3: THE SELECTION NAMESPACE -----------------------------------
#
# NO SUPERSESSION HEAD CHECK HERE, deliberately: no selection record carries a
# `supersedes` field, so there is no head to derive. The sequence and the
# predecessor's bytes are what the store can actually answer.

  || { echo "REFUSE: CSEL-000004 is not the accepted record; it must be immutable"; exit 1; }

             /var/lib/kyri/fabric/capability-selections/*.yaml 2>/dev/null)" \
  || { echo "REFUSE: a selection record now carries a supersedes field; this block's reasoning about selection heads no longer holds and must be re-reviewed"; exit 1; }

  || { echo "REFUSE: /etc/kyri/fabric/croute-0007.json is absent; CSEL-000005 resolves through CROUTE-0007 and is frozen against it"; exit 1; }

for pair in "the STALE G11-BC-AM selection body, recorded 2026-10-06T06:26:30-05:00 -- before CINST-000007 was admitted AND before CROUTE-0007 was decided -- and the same 606 bytes as this one:${STALE_AM_SELECTION_BODY}" \
            "the CSEL-000003 input:${ACCEPTED_CSEL3}" \
            "the CSEL-000004 input, which selects through the SUPERSEDED CROUTE-0006:${ACCEPTED_CSEL4}"; do

  echo "REFUSE: the body carries request_id ${BODY_REQUEST_ID}, reviewed ${REVIEWED_REQUEST_ID}"
  rm -f "${TMP}"; exit 1; }
test "${BODY_RECORDED}" = "${REVIEWED_RECORDED_AT}" || {
  echo "REFUSE: the body carries recorded_at ${BODY_RECORDED}, reviewed ${REVIEWED_RECORDED_AT}"
  rm -f "${TMP}"; exit 1; }
test "${BODY_EVALUATED}" = "${REVIEWED_RECORDED_AT}" || {
  echo "REFUSE: the body carries evaluated_at ${BODY_EVALUATED}, reviewed ${REVIEWED_RECORDED_AT}"

  echo "REFUSE: the body carries selected_at, which is not an input member -- the"
  echo "        engine derives it from recorded_at and would refuse this body with"
  echo "        'the decision body does not match this operation'"
  rm -f "${TMP}"; exit 1; }
printf 'ok  reviewed body %s, %s bytes\n' "${ACTUAL}" "${REVIEWED_BYTES}"
printf 'ok  request_id %s\n' "${BODY_REQUEST_ID}"
printf 'ok  recorded_at and evaluated_at both %s, and selected_at is left to the engine\n' \
  "${BODY_RECORDED}"

# ---- GATE 5: THE TEMPORAL GATE -- CEREMONY AUTHORITY -------------------
#
# THE ENGINE ENFORCES NONE OF THIS (G11-BC-AS: eight probes, `would_accept: true`
# in every one, including a selection dated a day before its route and one dated
# in December). Every bound is read out of the record it guards.
printf -- '\n--- the temporal gate: ordering the engine will not enforce ---\n'

        /var/lib/kyri/fabric/capability-routes/CROUTE-0007.yaml \
        /var/lib/kyri/fabric/capability-instances/CINST-000007.yaml \
        /var/lib/kyri/fabric/capability-advertisements/CADV-000008.yaml <<'GATE_PY'
import json
import re
import sys
from datetime import datetime

body = json.load(open(sys.argv[1], encoding="utf-8"))
route = open(sys.argv[2], encoding="utf-8").read()
instance = open(sys.argv[3], encoding="utf-8").read()
advertisement = open(sys.argv[4], encoding="utf-8").read()
now = datetime.now().astimezone()

# The route's own decision instant lives under `evidence:`, indented, and the
# record also carries a date-only provenance.recorded_at. Anchor on the indent so
# the two cannot be confused.
route_recorded = re.search(r"^  recorded_at: '(\d{4}-\d\d-\d\dT[^']+)'$",
                           route, re.M).group(1)


def field(text, name):
    return re.search(rf"^{name}: '(.+)'$", text, re.M).group(1)


admitted = field(instance, "admitted_at")
admitted_until = field(instance, "admitted_until")
adv_observed = field(advertisement, "observed_at")
adv_until = field(advertisement, "valid_until")
recorded = body["recorded_at"]
evaluated = body["evaluated_at"]

print(f"now                        {now.isoformat()}")
print(f"CADV-000008 valid_until    {adv_until}")
print(f"CINST-000007 admitted_at   {admitted}")
print(f"CINST-000007 until         {admitted_until}")
print(f"CROUTE-0007 recorded_at    {route_recorded}")
print(f"CSEL-000005 recorded_at    {recorded}")
print(f"CSEL-000005 evaluated_at   {evaluated}")

r = datetime.fromisoformat(recorded)
e = datetime.fromisoformat(evaluated)

# 1. A selection cannot be decided before the route it resolves through.
if r < datetime.fromisoformat(route_recorded):
    print(f"REFUSE: the selection is recorded {recorded}, BEFORE the route it "
          f"resolves through was decided at {route_recorded}. The engine accepts "
          f"this and still resolves an instance; it is a decision made through a "
          f"route that did not yet exist.")
    sys.exit(1)
# 2. Nor before the admission it selects.
if e < datetime.fromisoformat(admitted) or r < datetime.fromisoformat(admitted):
    print(f"REFUSE: the selection instants ({recorded} / {evaluated}) fall before "
          f"the admission at {admitted}")
    sys.exit(1)
# 3. Nor at or after the admission expires.
if (r >= datetime.fromisoformat(admitted_until)
        or e >= datetime.fromisoformat(admitted_until)):
    print(f"REFUSE: the selection instants ({recorded} / {evaluated}) fall at or "
          f"after the admission expires {admitted_until}")
    sys.exit(1)
# 4. Nor at or after the governing advertisement expires. Stated separately from
#    the admission rather than looped, so the bound this gate enforces is
#    readable in this file and not assembled at runtime.
if (r >= datetime.fromisoformat(adv_until)
        or e >= datetime.fromisoformat(adv_until)):
    print(f"REFUSE: the selection instants ({recorded} / {evaluated}) fall at or "
          f"after the governing advertisement expires {adv_until}")
    sys.exit(1)
print("ok  route recorded_at <= selection instants < both expiries")

# 5. A decision cannot be in the future.
for name, instant in (("recorded_at", r), ("evaluated_at", e)):
    if instant > now:
        print(f"REFUSE: {name} {instant.isoformat()} is in the FUTURE. The body "
              f"would claim a decision nobody has made.")
        sys.exit(1)
print("ok  both selection instants are in the past and already reached")

# 6. And the ceremony itself must run inside both governing windows.
for label, opens, closes in (("the advertisement", adv_observed, adv_until),
                             ("the admission", admitted, admitted_until)):
    if now < datetime.fromisoformat(opens):
        print(f"REFUSE: {label} window has not opened")
        sys.exit(1)
    if now >= datetime.fromisoformat(closes):
        print(f"REFUSE: {label} window has CLOSED at the current clock")
        print("REFUSE: do not extend it -- RE-PREPARE the chain from a fresh observation")
        sys.exit(1)
print("ok  the current clock is inside BOTH governing windows")
GATE_PY
)" || { printf '%s\n' "${GATE}"; echo "REFUSE: the temporal gate failed"; rm -f "${TMP}"; exit 1; }
printf '%s\n' "${GATE}"

# ---- GATE 6: THE RESOLUTION, WHICH IS THE WHOLE POINT ------------------

# THE MOST IMPORTANT GATE IN THIS FILE. `would_accept` is NOT the test: the

# selects nothing is exactly what this chain was renewed to stop producing.
printf -- '\n--- the resolution: it must be CINST-000007, not merely acceptable ---\n'
STAGE="$(mktemp -d)"

chmod 0400 "${STAGE}/csel-000005.json"
PREFLIGHT="$(python3 -m tools.fabric.cli select \
  --store-root /var/lib/kyri/fabric --expected-uid 1000 --expected-gid 1000 \
  --input-file csel-000005.json --approved-directory "${STAGE}" \
  --trust-store-root /var/lib/kyri/trust --preflight)"
rm -rf "${STAGE}"
printf '%s\n' "${PREFLIGHT}"

read -r ACCEPT MUTATED EXISTS PREDICTED SELECTED DIGEST PID <<<"$(printf '%s' "${PREFLIGHT}" \
  | python3 -c 'import json,sys; d=json.load(sys.stdin); print(d["would_accept"], d["mutated"], d["destination_exists"], d["predicted_record_id"], d["selected_instance_id"], d["request_digest"], d["request_id"])')"

# The resolution first, because it is the one that matters.
test "${SELECTED}" != "None" || {
  echo "REFUSE: the selection resolves to NULL. The engine would still accept it."
  echo "        An accepted selection that selects nothing is the quiet failure"
  echo "        this renewal chain exists to repair. Do not write it."
  rm -f "${TMP}"; exit 1; }
test "${SELECTED}" = "${REQUIRED_SELECTED_INSTANCE}" || {
  echo "REFUSE: the selection resolves to ${SELECTED}, not ${REQUIRED_SELECTED_INSTANCE}"
  rm -f "${TMP}"; exit 1; }
test "${ACCEPT}" = "True"   || { echo "REFUSE: would_accept is ${ACCEPT}"; exit 1; }
test "${MUTATED}" = "False" || { echo "REFUSE: the preflight reports mutated=${MUTATED}"; exit 1; }
test "${EXISTS}" = "False"  || { echo "REFUSE: the destination already exists"; exit 1; }
test "${PREDICTED}" = "CSEL-000005" \
  || { echo "REFUSE: the preflight predicts ${PREDICTED}, not CSEL-000005"; exit 1; }
test "${DIGEST}" = "${REQUEST_DIGEST}" \
  || { echo "REFUSE: the request digest is ${DIGEST}, not the reviewed ${REQUEST_DIGEST}"; exit 1; }
test "${PID}" = "${REVIEWED_REQUEST_ID}" \
  || { echo "REFUSE: the preflight reports request_id ${PID}, not the reviewed ${REVIEWED_REQUEST_ID}"; exit 1; }
printf 'ok  selected_instance_id %s  <-- the gate that matters\n' "${SELECTED}"
printf 'ok  would_accept %s, mutated %s, destination_exists %s, predicted %s\n' \
  "${ACCEPT}" "${MUTATED}" "${EXISTS}" "${PREDICTED}"
printf 'ok  request_digest %s\n' "${DIGEST}"
printf 'ok  request_id     %s\n' "${PID}"

printf -- '\n--- production Fabric must be byte-identical after the preflight ---\n'
test "$(aggregate /var/lib/kyri/fabric)" = "${FABRIC_BEFORE}" \
  || { echo "REFUSE: /var/lib/kyri/fabric changed during the preflight"; exit 1; }
test "$(cat /var/lib/kyri/fabric/sequences/capability-selection.seq)" = "4" \
  || { echo "REFUSE: the selection sequence moved during the preflight"; exit 1; }
printf 'ok  fabric unchanged at %s, selection sequence still 4\n' "${FABRIC_BEFORE}"

# ---- GATE 7: CURRENT AUTHORITY, ON A COPY CARRYING THE SELECTION -------
#
# What the renewal is FOR. The released verifier is asked the question the
# capability runtime asks before it executes anything, at the real clock, against
# a copy that carries the candidate. Against production as it stands today the
# same verifier answers `admission-window-not-open`.
printf -- '\n--- current execution authority, on a copy carrying the candidate ---\n'
SCRATCH="$(mktemp -d -p /data/kyri g11bcas-authority.XXXXXX)"
cp -a /var/lib/kyri/fabric "${SCRATCH}/fabric"
