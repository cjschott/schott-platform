#!/usr/bin/env bash
# CINV-000003 STAGE 1, exactly as it is to be typed.
#
# This file is reference text, not a script to run: the operator pastes it into
# a shell so every command and every refusal is visible on the console.
#
# =====================================================================
# STAGE 1 IS THE FIRST IRREVERSIBLE STEP
# =====================================================================
#
# Stage 0 created only a work area, outside every governed store, and could be
# abandoned at no cost. This is different. `capability invoke` ALLOCATES
# CINV-000003, writes an immutable invocation record and advances
# capability-invocation.seq from 2 to 3. None of that can be taken back:
# invocation identities are spent, not reserved, and Fabric and the Capability
# Runtime are append-only.
#
# THE IDENTITY IS SPENT EVEN IF THE INVOCATION IS REFUSED. `record_invocation`
# allocates the CINV inside its critical section BEFORE it decides whether the
# evidence supports the invocation, and on a refusal it allocates a CRES too
# and writes a refusal result. So a Stage 1 that refuses still consumes
# CINV-000003 and would consume CRES-000002. There is no path through this
# block that allocates nothing and still reaches the invoke. That is why every
# gate below runs first, and why they are worth their cost here in a way they
# were not at Stage 0.
#
# =====================================================================
# rc=1 IS SUCCESS. THE JSON IS THE VERDICT.
# =====================================================================
#
# `command_invoke` ends with an unconditional `return EXIT_DENIED`, under the
# comment "Every outcome reachable here is a governed negative", and
# EXIT_DENIED is 1. `command_invoke` supplies neither an adapter nor an
# execution binding, so `_bound_adapter_identity(None, None)` is None, nothing
# can execute, and `prepare_invocation` returns `prepared` with
# `no_authorised_adapter`.
#
# So a fully successful Stage 1 writes an immutable CINV and exits 1. The exit
# code is not the verdict and must not be read as one -- but neither is it
# ignored: this block requires rc to be EXACTLY 1, because rc=0 from this
# command would mean the released contract had changed underneath the ceremony.
#
# THE INVOKE IS NOT RUN UNDER `set -e`. The block sets `set +e` immediately
# around it and restores `set -e` after capturing the status. Under `set -e` an
# expected rc=1 would kill the shell after the record was written and before
# anything verified it, which is the worst possible moment to lose control of
# the ceremony.
#
# =====================================================================
# WHAT STAGE 1 MAY AND MAY NOT DO
# =====================================================================
#
# May, and exactly this -- measured by rehearsing the real write against a copy
# of the production runtime store and diffing the file list:
#
#   create  capability-invocations/CINV-000003.yaml      (one file, mode 0600)
#   change  sequences/capability-invocation.seq  2 -> 3
#
# Nothing else. The staged package tree is REUSED, not rebuilt: a tree already
# published under the same commitment is re-inspected in full and adopted, so
# staging/ does not change. No lock or journal file survives the critical
# section.
#
# May not: allocate a CRES on the success path, authorise a launch, execute
# anything, start a container, or touch Fabric. The flow ends before any
# surface that could run anything.
#
# RECOVERY AFTER STAGE 1. An invocation with a durable record and no terminal
# result is the honest interrupted state, and it is exactly what `capability
# recover` enumerates. It must NOT be "closed by hand": doing so destroys the
# evidence the enumeration acts on. If this block refuses after the invoke has
# run, stop and report -- do not re-run it. A second invoke with the same
# invocation_id resolves to the existing CINV and returns `consumed`, which is
# a different and non-idempotent outcome.
set -Eeuo pipefail
cd /opt/schott-platform

# =====================================================================
# BLOCK A -- CONTAINER AND IMAGE OBSERVATION. READ-ONLY. RUN THIS FIRST.
# =====================================================================
#
# This needs the kyri-capability account, and a sudo prompt must not land in
# the middle of the irreversible block. So it runs on its own, and records what
# it found in a witness file that BLOCK B requires and re-checks.
#
# `runuser` inherits the caller's working directory, and kyri-capability cannot
# traverse /opt/schott-platform -- which is why the Stage 0 observation failed
# the first time it was run. It is run from /tmp for that reason. Nothing
# governed changed then and nothing changes here.
#
# Do NOT pull, build, load or retag anything.
bash <<'OBSERVE'
set -Eeuo pipefail
WITNESS=/data/kyri/work/g11bcn-stage-1-witness
EXPECTED_IMAGE=5cee2b5305b5c5ebe3e8f4facfd1a6cc2c2057a7d301d6869783dddc463f5190
TARGET_CONTAINER=kyri-CINV-000003

cd /tmp

printf '\n--- containers ---\n'
CONTAINERS="$(sudo runuser -u kyri-capability -- env HOME=/data/kyri/capability \
  XDG_RUNTIME_DIR=/run/user/999 podman ps -a --format '{{.Names}} {{.Status}}')"
printf '%s\n' "${CONTAINERS}"

# No container for the identity this ceremony is about to allocate, and no
# kyri-CINV-* container at all: a stray one means an execution plane that is
# not in the state this ceremony assumes.
if printf '%s' "${CONTAINERS}" | grep -q "${TARGET_CONTAINER}"; then
  echo "REFUSE: a ${TARGET_CONTAINER} container already exists"
  exit 1
fi
if printf '%s' "${CONTAINERS}" | grep -q 'kyri-CINV-'; then
  echo "REFUSE: a kyri-CINV-* container exists; the execution plane is not idle"
  exit 1
fi
printf 'ok  no %s container, no kyri-CINV-* container\n' "${TARGET_CONTAINER}"

printf '\n--- images ---\n'
IMAGES="$(sudo runuser -u kyri-capability -- env HOME=/data/kyri/capability \
  XDG_RUNTIME_DIR=/run/user/999 podman images --no-trunc --format '{{.ID}} {{.Repository}}:{{.Tag}}')"
printf '%s\n' "${IMAGES}"

printf '%s' "${IMAGES}" | grep -q "${EXPECTED_IMAGE}" \
  || { echo "REFUSE: the execution image ${EXPECTED_IMAGE} is not present"; exit 1; }
printf '%s' "${IMAGES}" | grep -q 'localhost/kyri-capability-execution:g5' \
  || { echo "REFUSE: localhost/kyri-capability-execution:g5 is not present"; exit 1; }
printf 'ok  execution image %s present\n' "${EXPECTED_IMAGE}"

# The witness. BLOCK B requires it, requires it to be fresh, and requires it to
# say exactly this. It is not a substitute for the observation -- it is a record
# that the observation was made and what it concluded.
umask 077
printf 'stage-1-observation\nimage %s\nno-target-container %s\n' \
  "${EXPECTED_IMAGE}" "${TARGET_CONTAINER}" > "${WITNESS}"
printf '\nok  witness written to %s\n' "${WITNESS}"
OBSERVE

import hashlib
import json
import pathlib
import sys

from tools.capability.invocation_identity import canonical_bytes


def refuse(message):
    print(f"REFUSE: {message}")
    sys.exit(1)


try:
    raw = pathlib.Path(sys.argv[1]).read_bytes()
    document = json.loads(raw)
except (OSError, ValueError) as exc:
    refuse(f"the payload is not readable JSON ({exc})")

try:
    blob = canonical_bytes(document)
except Exception as exc:  # noqa: BLE001
    refuse(f"the released canonicalizer refused the payload ({exc})")

digest = hashlib.sha256(blob).hexdigest()
expected_digest = sys.argv[2]
expected_bytes = int(sys.argv[3])

print(f"canonical  {digest}  ({len(blob)} bytes)")

if digest != expected_digest:
    refuse(f"canonical digest {digest}, reviewed {expected_digest}")
if len(blob) != expected_bytes:
    refuse(f"canonical bytes {len(blob)}, reviewed {expected_bytes}")

    refuse(f"the payload operation is {document.get('operation')}, reviewed verify-execution-boundary")
arguments = document.get("arguments")
