aggregate() { find "$1" -type f -print0 | sort -z | xargs -0 sha256sum | sha256sum | cut -d' ' -f1; }

#
# G11-BC-W's lesson, kept: an operation must run against the runtime that can
# read what it writes. A Generation-19 runtime has no correct-provenance verb
# at all, so this would fail loudly rather than quietly -- but "loudly" is not
# a reason to skip the check.
printf -- '--- installed generation gate ---\n'
check_installed() {
  local relative="$1" wanted="$2" got
  got="$(sha256sum "${INSTALLED}/${relative}" 2>/dev/null | cut -d' ' -f1)"
  test "${got}" = "${wanted}" \
    || { echo "REFUSE: ${INSTALLED}/${relative} is ${got:-absent}, not the reviewed Generation-20 ${wanted}"; exit 1; }
  printf 'ok  %-28s %s\n' "$(basename "${relative}")" "${got}"
}

check_installed tools/capability/execution/admin.py         f691f914058491b1e7ccb3dd8498a667588a4fe36ffee13b0777733617845606

check_installed tools/capability/execution/provenance.py    2783c5438f1154111dc3700b6b9da74f54b3a71d7ec9fc12f17585bc01ab0bd6

check_sha() {
  local path="$1" wanted="$2" got
  got="$(sha256sum "${path}" 2>/dev/null | cut -d' ' -f1)"
  test -n "${got}" || { echo "REFUSE: ${path} is absent"; exit 1; }
  test "${got}" = "${wanted}" \
    || { echo "REFUSE: ${path} is ${got}, expected ${wanted}"; exit 1; }
  printf 'ok  %-58s %s\n' "$(basename "${path}")" "${got}"
}
