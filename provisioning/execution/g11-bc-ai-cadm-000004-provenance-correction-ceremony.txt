# This file is reference text, not a script to run: the operator pastes it into

# =====================================================================
# DO NOT RUN THIS UNTIL BOTH ARE TRUE
# =====================================================================
#   1. Generation 22 is INSTALLED and accepted.
#   2. The reviewer has accepted ADR-0018 and this ceremony.
#
# Gate 0 below refuses if (1) is not true. Nothing can check (2) but you.
#
# WHAT THIS DOES, IN ONE SENTENCE
# ===============================
# It records that all three of CADM-000004's authority assertions -- `actor`,
# `request_id` and `recorded_at` -- are not truthful provenance, while leaving
# CADM-000004, the lifecycle, the occupancy and every other record exactly as
# they are.
#
# THREE CORRECTIONS, NOT ONE
# ==========================
# One correction record disputes one claim. Three false claims therefore take
# three records, and that is the design rather than a limitation: a single record
# disputing three fields could not be read back as "which claim is disputed, and
# on what evidence", and a later dispute of a fourth claim would have nowhere to
# go. Each is create-once per (subject, field), so re-running this ceremony after
# a partial run RESUMES -- it reports the records that already exist and writes
# only the ones that do not.
#
# WHAT IT IS NOT
# ==============
# It is NOT ratification. It does not say an operator approved the abandonment,
# it names no substitute actor, and it makes no claim that the original action
# was correct to take. It is NOT a reversal: CINV-000001 stays `abandoned`, the
# released slot stays released, and no transition is written.
#
# IT DOES NOT DISPUTE `reason`. `historical-incomplete-execution` stands, and
# that is a decision, not an omission. The abandonment validated that category
# against the store before writing -- CINV-000001 has no terminal result -- so
# the category is a FACT about what happened, not an assertion about who claimed
# it. `reason` is not in CORRECTABLE_FIELDS and the verb would refuse it.
#
# WHY THE STORE IS IN THIS STATE
# ==============================
# On 2026-09-24 at 06:40:43-05:00 a test sweep ran
# tests/test-capability-execution-generation20-installer.sh. On a
# Generation-21 host that suite's fixture left a Generation-19 `cli.py` in place
# and then dispatched `abandon --cinv CINV-000001 --actor x --request-id y` with
# NO `--store-root`, so the compiled-in production root resolved and CADM-000004
# was written against production.
#
# The reviewer RETAINED the lifecycle effect -- CINV-000001 had no terminal
# result and `historical-incomplete-execution` is the category the store admits
# -- and REJECTED the asserted authority. So the effect is truthful and the
# three assertions are not:
#
#   field        disputed value               finding (ADR-0018)
#   -----------  ---------------------------  -------------------------------
#   actor        x                            attribution-not-authorised
#   request_id   y                            assertion-synthetic
#   recorded_at  2026-09-20T20:00:00-05:00    assertion-synthetic
#
# THE FINDING IS PER FIELD, AND THE TWO ARE NOT INTERCHANGEABLE.
#
#   `attribution-not-authorised` says: this record claims an ACTOR, and that
#   actor did not authorise the action. `actor` is the field whose whole job is
#   to name who acted, so a false one is an unauthorised attribution whatever
#   literal it holds -- `x` is a claim about an actor, and the claim is not
#   authorised.
#
#   `assertion-synthetic` says: the recorded value was never asserted by ANY
#   authority -- a literal that reached the record through a mechanism with no
#   standing to assert anything. `y` is that, and so is a timestamp copied out
#   of a test.
#
# ADR-0018 fixes the mapping in a table, and the ceremony reads it from a table
# for the same reason: a single global finding applied to all three -- which is
# what this ceremony carried when it was first prepared at G11-BC-AI -- would
# record `actor` under the wrong category, and a correction recorded under the
# wrong finding is itself untruthful provenance.
#
# THE THIRD INSTANT, AND WHY IT IS SEPARATE
# =========================================
# `--actual-occurrence-at 2026-09-24T06:40:43-05:00` is when the abandonment
# observably happened, taken from the mtime of the record the escape wrote and
# from the sweep's own output. `--recorded-at` is when THIS correction is being
# written. They are different events and neither is the disputed value, so there
# are three instants and they have three names. `--actual-occurrence-at` asserts
# no authority: it is an observation, and it is the reason the backdating can be
# recorded as backdating rather than merely as "wrong".
#
# THE TARGET IS NAMED, AND THE WRITER IS ASKED WHERE IT WROTE
# ===========================================================
# `--store-root /data/kyri/capability-runtime` is spelled out below, and the
# emitted `target` is checked against the inode this ceremony measured itself.
# That check exists because of this very incident: everything the escaped suite
# checked was a check about TEXT, and the one thing that mattered was which
# object the writer held.

# ONE `--recorded-at` FOR ALL THREE. They are one operator act, recorded at one
# instant, and giving each its own `date -Is` would put three different times on
# one decision. `--actual-occurrence-at` is the same for all three too, and for a

correct_one() {
  local field="$1" value="$2" finding="$3" expected_cadm="$4"
  local out="${GATE_DIR}/${field}.json"
  printf -- '\n--- capability correct-provenance %s (production, installed runtime) ---\n' "${field}"
  set +e
  ( cd "${INSTALLED}" && python3 -m tools.capability.cli correct-provenance \
      --store-root "${RUNTIME}" \
      --subject-cadm "${SUBJECT}" \
      --cinv "${TARGET_CINV}" \
      --disputed-field "${field}" \
      --disputed-value "${value}" \
      --finding "${finding}" \
      --actual-initiator "${INITIATOR}" \
      --actor "${ACTOR}" \
      --request-id "g11bcai-correct-cadm-000004-${field}" \
      --recorded-at "${RECORDED_AT}" \
      --actual-occurrence-at "${OCCURRED_AT}" \
      --evidence-reference "${INCIDENT_REPORT}" \
      --evidence-reference "${ARCHITECTURE}" \
      --evidence-reference "${HARNESS}" ) > "${out}"
  local rc=$?
  set -e
  cat "${out}"
  test "${rc}" = "0" \
    || { echo "STOP: the process exited ${rc}. Do not run the remaining fields. Report the output."; exit 1; }

  local got
  verdict() {
    got="$(python3 -c 'import json,sys; print(json.load(open(sys.argv[1]))[sys.argv[2]])' "${out}" "$1")"
    test "${got}" = "$2" || { echo "STOP: $1 is ${got}, expected $2"; exit 1; }
    printf 'ok  %-22s %s\n' "$1" "${got}"
  }
  verdict cadm                 "${expected_cadm}"
  verdict subject_cadm         "${SUBJECT}"
  verdict subject_member       abandonment
  verdict subject_digest       "${CADM4_ABANDONMENT_SHA}"
  verdict cinv                 "${TARGET_CINV}"
  verdict disputed_field       "${field}"
  verdict disputed_value       "${value}"
  verdict finding              "${finding}"
  verdict actual_initiator     "${INITIATOR}"
  verdict actual_occurrence_at "${OCCURRED_AT}"
  verdict effect               retained
  verdict lifecycle_state      abandoned
  verdict resumed              False
  verdict store_root           "${RUNTIME}"

  # WHERE IT WROTE, not where it was told to write.
  local wrote_inode wrote_dev
  wrote_inode="$(python3 -c 'import json,sys; print(json.load(open(sys.argv[1]))["target"]["st_ino"])' "${out}")"
  wrote_dev="$(python3 -c 'import json,sys; print(json.load(open(sys.argv[1]))["target"]["st_dev"])' "${out}")"
  if [[ "${wrote_inode}" != "${EXECUTION_INODE}" || "${wrote_dev}" != "${EXECUTION_DEV}" ]]; then
    echo "STOP: the writer held device ${wrote_dev} inode ${wrote_inode}, not the production execution root ${EXECUTION_DEV}/${EXECUTION_INODE}"
    exit 1
  fi
  printf 'ok  the writer held the production execution root, asked of the kernel\n'

  # The three instants are three, IN THE DURABLE RECORD -- not in the emitted
  # summary, which does not carry `recorded_at` at all. What was written is the
  # thing that has to be right.
  python3 - "${RUNTIME}/execution/admin-records/${expected_cadm}/provenance-correction" <<'INSTANTS_PY' || exit 1
import json
import sys

document = json.load(open(sys.argv[1], encoding="utf-8"))
occurrence = document["actual_occurrence_at"]
recorded = document["recorded_at"]
if occurrence == recorded:
    print("STOP: the occurrence and the recording are the same instant")
    sys.exit(1)
print("ok  the occurrence is not the recording")
if document["disputed_field"] == "recorded_at":
    disputed = document["disputed_value"]
    if len({disputed, occurrence, recorded}) != 3:
        print(f"STOP: the three instants are not three: disputed={disputed} "
              f"occurrence={occurrence} recorded={recorded}")
        sys.exit(1)
    print("ok  three instants, three distinct values, in the written record")
if document.get("action_reversed") is not False:
    print("STOP: the record does not state that nothing was reversed")
    sys.exit(1)
if document.get("lifecycle_unchanged") is not True:
    print("STOP: the record does not state that the lifecycle did not move")
    sys.exit(1)
print("ok  the record states action_reversed false and lifecycle_unchanged true")
INSTANTS_PY
}
